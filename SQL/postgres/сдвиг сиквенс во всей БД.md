```sql
DO $$
DECLARE
    rec record;
    max_id bigint;
    processed_seqs text[] := '{}'; -- Массив для защиты от дубликатов
BEGIN
    -- Ищем все колонки с nextval, используя системные каталоги (надежнее, чем information_schema)
    FOR rec IN
        SELECT DISTINCT
            n.nspname AS table_schema,
            c.relname AS table_name,
            a.attname AS column_name,
            pg_get_serial_sequence(quote_ident(n.nspname) || '.' || quote_ident(c.relname), a.attname) AS seq_name
        FROM pg_class c
        JOIN pg_namespace n ON n.oid = c.relnamespace
        JOIN pg_attribute a ON a.attrelid = c.oid
        WHERE c.relkind = 'r' -- Только обычные таблицы (исключаем view и matview)
          AND n.nspname NOT IN ('pg_catalog', 'information_schema')
          AND a.attnum > 0
          AND NOT a.attisdropped
          AND pg_get_serial_sequence(quote_ident(n.nspname) || '.' || quote_ident(c.relname), a.attname) IS NOT NULL
    LOOP
        -- Пропускаем, если этот секвенс уже был обработан (защита от наследования таблиц)
        IF rec.seq_name = ANY(processed_seqs) THEN
            CONTINUE;
        END IF;
        
        processed_seqs := array_append(processed_seqs, rec.seq_name);

        -- Получаем максимум
        EXECUTE format(
            'SELECT COALESCE(MAX(%I), 0) FROM %I.%I', 
            rec.column_name, rec.table_schema, rec.table_name
        ) INTO max_id;

        -- ГЛАВНОЕ ИСПРАВЛЕНИЕ: если таблица пуста (max_id = 0), не трогаем секвенс.
        -- Он и так находится на стартовом значении (обычно 1).
        IF max_id > 0 THEN
            EXECUTE format('SELECT setval(%L::regclass, %s)', rec.seq_name, max_id);
            RAISE NOTICE '✅ Синхронизирована: % (таблица %.%, макс. ID: %)', 
                         rec.seq_name, rec.table_schema, rec.table_name, max_id;
        ELSE
            RAISE NOTICE '⏭️ Пропущена (таблица пуста): % (таблица %.%)', 
                         rec.seq_name, rec.table_schema, rec.table_name;
        END IF;
    END LOOP;
    
    RAISE NOTICE '🎉 Синхронизация всех последовательностей завершена!';
END $$;
```
